cmake_minimum_required(VERSION 3.24)
project(Pizzeria LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ENABLE_COVERAGE "Collect GCC coverage" OFF)
option(STATIC_RELEASE "Link standalone Linux binaries" OFF)
if(STATIC_RELEASE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_link_options(-static)
endif()
if(ENABLE_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Coverage requires GCC")
  endif()
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
endif()
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
file(GLOB_RECURSE domain_sources CONFIGURE_DEPENDS src/*.cpp)
add_library(pizzeria_domain ${domain_sources})
foreach(group IN ITEMS stock menu sales kitchen service exceptions)
  target_include_directories(pizzeria_domain PUBLIC "include/${group}")
endforeach()
add_executable(pizzeria cli/main.cpp cli/Console.cpp)
target_link_libraries(pizzeria PRIVATE pizzeria_domain)
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(doctest
    URL https://codeload.github.com/doctest/doctest/tar.gz/refs/tags/v2.4.11
    URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    SOURCE_SUBDIR single-header-only)
  FetchContent_MakeAvailable(doctest)
  add_executable(pizzeria_tests tests/DomainTests.cpp tests/ExtendedTests.cpp)
  target_include_directories(pizzeria_tests SYSTEM PRIVATE "${doctest_SOURCE_DIR}")
  target_link_libraries(pizzeria_tests PRIVATE pizzeria_domain)
  add_test(NAME domain COMMAND pizzeria_tests)
  add_test(NAME model COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/CheckModel.cmake")
  add_test(NAME cli COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:pizzeria>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/RunCli.cmake")
  set_tests_properties(domain cli PROPERTIES TIMEOUT 20)
endif()
install(TARGETS pizzeria RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
find_program(DOXYGEN_EXECUTABLE doxygen)
if(DOXYGEN_EXECUTABLE)
  file(RELATIVE_PATH DOCS_SOURCE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
  configure_file(Doxyfile.in Doxyfile @ONLY)
  add_custom_target(docs COMMAND "${DOXYGEN_EXECUTABLE}" Doxyfile
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" VERBATIM)
endif()
